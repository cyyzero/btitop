cmake_minimum_required(VERSION 3.20)
project(btitop VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBBPF REQUIRED IMPORTED_TARGET libbpf)
pkg_check_modules(NCURSES REQUIRED IMPORTED_TARGET ncursesw)
find_package(Threads REQUIRED)
include(GNUInstallDirs)
find_program(CLANG_EXECUTABLE NAMES clang-18 clang REQUIRED)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
 set(BPF_ARCH arm64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64)$")
 set(BPF_ARCH x86)
else()
 message(FATAL_ERROR "Unsupported BPF architecture: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
file(GLOB BPFTOOL_CANDIDATES /usr/lib/linux-tools/*/bpftool)
list(REVERSE BPFTOOL_CANDIDATES)
if(BPFTOOL_CANDIDATES)
 list(GET BPFTOOL_CANDIDATES 0 DEFAULT_BPFTOOL)
else()
 find_program(DEFAULT_BPFTOOL NAMES bpftool REQUIRED)
endif()
set(BPFTOOL_EXECUTABLE ${DEFAULT_BPFTOOL} CACHE FILEPATH "bpftool binary")
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/vmlinux.h
 COMMAND ${BPFTOOL_EXECUTABLE} btf dump file /sys/kernel/btf/vmlinux format c > ${CMAKE_CURRENT_BINARY_DIR}/vmlinux.h
 DEPENDS /sys/kernel/btf/vmlinux)
set(BPF_OBJECT ${CMAKE_CURRENT_BINARY_DIR}/task_iter.bpf.o)
add_custom_command(OUTPUT ${BPF_OBJECT}
 COMMAND ${CLANG_EXECUTABLE} -O2 -g -Wno-missing-declarations -target bpf -D__TARGET_ARCH_${BPF_ARCH} -D__BPF__ -I${CMAKE_CURRENT_BINARY_DIR} -I${CMAKE_CURRENT_SOURCE_DIR}/include -c ${CMAKE_CURRENT_SOURCE_DIR}/bpf/task_iter.bpf.c -o ${BPF_OBJECT}
 DEPENDS bpf/task_iter.bpf.c ${CMAKE_CURRENT_BINARY_DIR}/vmlinux.h include/btitop/wire.h)
add_custom_target(bpf_program ALL DEPENDS ${BPF_OBJECT})
add_library(btitop_core src/core.cpp src/proc.cpp src/bpf.cpp src/system.cpp)
target_include_directories(btitop_core PUBLIC include)
target_compile_definitions(btitop_core PRIVATE BTITOP_BPF_OBJECT="${BPF_OBJECT}" BTITOP_INSTALLED_BPF_OBJECT="${CMAKE_INSTALL_FULL_LIBDIR}/btitop/task_iter.bpf.o")
target_link_libraries(btitop_core PUBLIC PkgConfig::LIBBPF Threads::Threads)
add_dependencies(btitop_core bpf_program)
add_executable(btitop src/main.cpp src/ui.cpp)
target_link_libraries(btitop PRIVATE btitop_core PkgConfig::NCURSES)
enable_testing()
add_executable(btitop-tests tests/test_core.cpp)
target_link_libraries(btitop-tests PRIVATE btitop_core)
add_test(NAME core COMMAND btitop-tests)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_integration.py $<TARGET_FILE:btitop>)

install(TARGETS btitop RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES ${BPF_OBJECT} DESTINATION ${CMAKE_INSTALL_LIBDIR}/btitop)
